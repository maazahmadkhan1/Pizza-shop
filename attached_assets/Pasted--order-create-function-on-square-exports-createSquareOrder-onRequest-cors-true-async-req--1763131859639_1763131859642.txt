// order create function on square
exports.createSquareOrder = onRequest({ cors: true }, async (req, res) => {
  try {
    if (req.method !== "POST") {
      return res.status(405).json({ error: "Method not allowed" });
    }

    const payload = req.body;
    const {
      orderNumber,
      customer,
      items,
      deliveryFee,
    } = payload;

    if (!customer?.squareCustomerId) {
      return res.status(400).json({ error: "Missing squareCustomerId" });
    }

    const squareAccessToken =
      process.env.SQUARE_ACCESS_TOKEN ||
      "EAAAl0xt86AQRcaAChDVWQC8LC7P_tYhekGXzpkAdEUX1odnXgmtmZ6wEqZB-pzl";

    const squareApiUrl = "https://connect.squareupsandbox.com/v2/orders";

    // LINE ITEMS
    const lineItems = items.map((item) => ({
      name: item.name,
      quantity: String(item.quantity),
      base_price_money: {
        amount: Math.round(item.price * 100),
        currency: "USD"
      }
    }));

    if (deliveryFee && deliveryFee > 0) {
      lineItems.push({
        name: "Delivery Fee",
        quantity: "1",
        base_price_money: {
          amount: Math.round(deliveryFee * 100),
          currency: "USD"
        }
      });
    }

    // CREATE ORDER (NO FULFILLMENT)
    const idempotencyKey = ${orderNumber.slice(-15)}_${Date.now()};
    const body = {
      order: {
        location_id: "LAJ4KFYABFMDA",
        customer_id: customer.squareCustomerId,
        reference_id: orderNumber,
        line_items: lineItems,
        state: "OPEN",
        fulfillments: []
      },
      idempotency_key: idempotencyKey
    };

    const response = await fetch(squareApiUrl, {
      method: "POST",
      headers: {
        Authorization: Bearer ${squareAccessToken},
        "Content-Type": "application/json",
        "Square-Version": "2025-01-15"
      },
      body: JSON.stringify(body)
    });

    const result = await response.json();
    if (!response.ok) throw new Error(result.errors?.[0]?.detail || "Order failed");

    const squareOrderId = result.order.id;

    // CALL INVOICE FUNCTION VIA INTERNAL URL
    let invoiceInfo = { invoiceId: null, invoiceUrl: null };
    try {
      const invoiceRes = await fetch(
        http://127.0.0.1:5001/${process.env.GCLOUD_PROJECT}/us-central1/createSquareInvoiceFromOrder,
        {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            squareOrderId,
            sendEmail: true,
            customer: {
              name: customer.name,
              email: customer.email,
              squareCustomerId: customer.squareCustomerId,
              phone: payload.deliveryAddress?.phone
            }
          })
        }
      );

      const invoiceData = await invoiceRes.json();
      if (invoiceRes.ok) {
        invoiceInfo = {
          invoiceId: invoiceData.invoiceId,
          invoiceUrl: invoiceData.publicUrl
        };
      } else {
        console.error("Invoice failed:", invoiceData);
      }
    } catch (err) {
      console.error("Invoice call failed:", err.message);
    }

    return res.status(200).json({
      success: true,
      message: "Order + Invoice created",
      squareOrderId,
      ...invoiceInfo,
      customerName: customer.name
    });

  } catch (error) {
    console.error("Error:", error);
    return res.status(500).json({ success: false, error: error.message });
  }
});

// INVOICE FUNCTION
exports.createSquareInvoiceFromOrder = onRequest({ cors: true }, async (req, res) => {
  try {
    if (req.method !== "POST") {
      return res.status(405).json({ error: "Method not allowed" });
    }

    const { squareOrderId, sendEmail = true, customer } = req.body;

    if (!squareOrderId || !customer?.squareCustomerId) {
      return res.status(400).json({ error: "Missing data" });
    }

    const squareAccessToken =
      process.env.SQUARE_ACCESS_TOKEN ||
      "EAAAl0xt86AQRcaAChDVWQC8LC7P_tYhekGXzpkAdEUX1odnXgmtmZ6wEqZB-pzl";

    const [givenName = "Customer", ...familyParts] = (customer.name || "").split(" ");
    const familyName = familyParts.join(" ");

    const invoiceBody = {
      invoice: {
        order_id: squareOrderId,
        location_id: "LAJ4KFYABFMDA",
        title: "Your Order Invoice",
        description: "Please pay to complete your order.",
        primary_recipient: {
          customer_id: customer.squareCustomerId
        },
        payment_requests: [{
          request_type: "BALANCE",
          due_date: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000).toISOString().split("T")[0]
        }],
        delivery_method: sendEmail ? "EMAIL" : "SHARE_MANUALLY",
        accepted_payment_methods: { card: true }
      },
      idempotency_key: ${squareOrderId.slice(-15)}_inv_${Date.now()}
    };

    const response = await fetch("https://connect.squareupsandbox.com/v2/invoices", {
      method: "POST",
      headers: {
        Authorization: Bearer ${squareAccessToken},
        "Content-Type": "application/json",
        "Square-Version": "2025-01-15"
      },
      body: JSON.stringify(invoiceBody)
    });

    const result = await response.json();
    if (!response.ok) throw new Error(result.errors?.[0]?.detail || "Invoice failed");

    const invoice = result.invoice;
    const publicUrl = https://squareup.com/pay-invoice/${invoice.public_id};

    return res.status(200).json({
      success: true,
      invoiceId: invoice.id,
      publicUrl,
      customerName: ${givenName} ${familyName}.trim()
    });

  } catch (error) {
    console.error("Invoice error:", error);
    return res.status(500).json({ success: false, error: error.message });
  }
});
